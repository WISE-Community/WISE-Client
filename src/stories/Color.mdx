# Color

WISE employs a color system based on [Material 3 static color schemes](https://m3.material.io/styles/color/static/baseline).
Colors are available as CSS custom properties (variables).
The main color role colors can also be applied to HTML elements using CSS classnames: <code>.primary</code>, <code>.secondary</code>, <code>.tertiary</code>, <code>.error</code>, <code>.success</code>.

Check out the [Angular Material theming guide](https://material.angular.dev/guide/theming) and [Material 3 color roles](https://m3.material.io/styles/color/roles) for more information.

export const ColorSwatch = ({ name, variable, text, showCode }) => {
  const [computedColor, setComputedColor] = useState('');
  const swatchRef = useRef(null);

  const rgbToHex = (rgb) => {
    if (!rgb || !rgb.startsWith('rgb')) return '';
    const [r, g, b] = rgb.match(/\d+/g).map(Number);
    return "#" + ((1 << 24) + (r << 16) + (g << 8) + b).toString(16).slice(1).toUpperCase();
  };

  useEffect(() => {
    if (swatchRef.current) {
      const color = getComputedStyle(swatchRef.current).backgroundColor;
      setComputedColor(rgbToHex(color));
    }
  }, [swatchRef]);

  const contrastColorVar = variable.includes('on-') ? variable.replace('on-', '') : `--mat-sys-on-${variable.substring(10)}`;
  return (
    <div
      ref={swatchRef}
      className="color"
      style={{
        backgroundColor: `var(${variable})`,
        color: text ? `var(${text})`: `var(${contrastColorVar})`,
      }}
    >
      <div>{name}</div>
      <div>
        <code>{computedColor}</code>
      </div>
      {showCode && (
        <div>
          <code>{variable}</code>
        </div>
      )}
    </div>
  );
};

export const ColorSwatchGroup = ({ description, name, showCode }) => {
  const capitalizedName = name.charAt(0).toUpperCase() + name.slice(1);
  const colors = [
    { name: capitalizedName, variable: `--mat-sys-${name}` },
    { name: `On ${capitalizedName}`, variable: `--mat-sys-on-${name}` },
    { name: `${capitalizedName} Container`, variable: `--mat-sys-${name}-container` },
    { name: `On ${capitalizedName} Container`, variable: `--mat-sys-on-${name}-container` },
  ];

  return (
    <div className="w-1/2 sm:w-1/3 flex flex-col gap-2 p-1">
      <div className="flex flex-col gap-2">
        <div>
          <ColorSwatch name={colors[0].name} variable={colors[0].variable} showCode={showCode} />
          <ColorSwatch name={colors[1].name} variable={colors[1].variable} showCode={showCode} />
        </div>
        <div>
          <ColorSwatch name={colors[2].name} variable={colors[2].variable} showCode={showCode} />
          <ColorSwatch name={colors[3].name} variable={colors[3].variable} showCode={showCode} />
        </div>
      </div>
      {description && <div className="text-sm mb-4">{capitalizedName}: {description}</div>}
    </div>
  );
};

export const ColorSwatchList = ({ colors, direction = 'horizontal', showCode }) => (
  <div>
    <div className={`color-list flex ${direction === 'vertical' ? 'flex-col' : 'flex-wrap'}`}>
      {colors.map(color => (
        <div style={{width: direction === 'horizontal' ? (100 / colors.length) + '%' : ''}}>
          <ColorSwatch key={color.variable} name={color.name} variable={color.variable} text={color.text} showCode={showCode} />
        </div>
      ))}
    </div>
  </div>
);
